一、代码块

粗体用更亮的一档灰，斜体保持原样，链接用蓝色：editor.example。

// One Dark Pro Darker
pub struct WorldBounds {
    pub min_y: i64,
    pub max_y: i64,
}

impl WorldBounds {
    pub fn new(min_y: i64, max_y: i64) -> Result {
        if min_y >= max_y {
            return Err("bounds");
        }
        Ok(Self { min_y, max_y })
    }
}
